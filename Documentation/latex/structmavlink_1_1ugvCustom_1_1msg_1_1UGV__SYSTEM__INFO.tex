\doxysubsection{Detailed Description}
\mbox{\hyperlink{structmavlink_1_1ugvCustom_1_1msg_1_1UGV__SYSTEM__INFO}{UGV\+\_\+\+SYSTEM\+\_\+\+INFO}} message. 

\begin{DoxyVerb}     Reports subsystem communication and health status of the UGV platform.
     Each subsystem uses 2-bit encoding:
     0x01 = No communication
     0x10 = Communicating and healthy
     0x11 = Communicating and unhealthy (fault present)
     0x00 = Reserved
\end{DoxyVerb}
