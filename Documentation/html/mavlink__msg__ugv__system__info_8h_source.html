16  Bit 6-7 : LV Battery Health
17  */

19  Bit 0-1 : LV PDU

22  Bit 6-7 : VCU
23  */
24  uint8_t subsystem_health_3; /*<
25  Bit 0-1 : Front Left Motor

27  Bit 4-5 : Front Right Motor
28  Bit 6-7 : Rear Right Motor
29  */
30  uint8_t subsystem_health_4; /*<
31  Bit 0-1 : UHF Radio
32  Bit 2-3 : LBAND Radio
33  Bit 4-5 : Compute
34  Bit 6-7 : Reserved
35  */
36  uint8_t battery_soc; /*<
37  soc percentage of remaining battery
38  */
39  uint8_t main_mode; /*<
40  Current active main mode.
41  */
42  uint8_t sub_mode; /*<
43  Current active sub mode.
44  */
45  uint8_t intended_main_mode; /*<
46  Last main mode commanded by GCS.
47  */
48  uint8_t intended_sub_mode; /*<
49  Last sub mode commanded by GCS.
50  */
51  uint8_t mode_change_reason; /*<
